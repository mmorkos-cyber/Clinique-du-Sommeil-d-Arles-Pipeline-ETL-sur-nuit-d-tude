CREATE DEFINER=`root`@`localhost` PROCEDURE `insert_data_resultat`(
IN p_id_nuit INT,
IN spo2_min DECIMAL (5,2),
IN spo2_moy DECIMAL (5,2),
IN spo2_mediane DECIMAL (5,2),
IN nb_ronflements_forts INT,
IN decibels_max DECIMAL (5,2),
IN decibels_moy DECIMAL (5,2),
IN position_dominante VARCHAR (20),
IN duree_hypoxie_min INT
)
BEGIN
INSERT INTO resultat_nuit(
    id_nuit,
    id_medecin_validateur,
    date_validation,
    iah,
    spo2_min,
    spo2_moy,
    spo2_mediane,
    nb_apnees,
    nb_hypopnees,
    nb_rera,
    nb_microeveils,
    duree_sommeil_min,
    duree_hypoxie_min,
    position_dominante,
    duree_apnee_moy_sec,
    duree_apnee_max_sec,
    decibels_max,
    decibels_moy,
    nb_ronflements_forts
)
VALUES (
    p_id_nuit,

    (SELECT id_medecin
     FROM nuit_etude
     WHERE id_nuit = p_id_nuit),

    CURDATE(),

    (SELECT COUNT(*)
    FROM evenement_respiratoire
    WHERE id_nuit = p_id_nuit
      AND type_evenement IN (
          'apnée obstructive',
          'apnée centrale',
          'hypopnée')
) / 7,

    spo2_min,
    spo2_moy,
    spo2_mediane,

    (SELECT COUNT(*)
     FROM evenement_respiratoire
     WHERE id_nuit = p_id_nuit
       AND type_evenement IN ('apnée obstructive','apnée centrale')),

    (SELECT COUNT(*)
     FROM evenement_respiratoire
     WHERE id_nuit = p_id_nuit
       AND type_evenement = 'hypopnée'),

    (SELECT COUNT(*)
     FROM evenement_respiratoire
     WHERE id_nuit = p_id_nuit
       AND type_evenement = 'RERA'),
       
	(SELECT COUNT(*)
     FROM evenement_respiratoire
     WHERE id_nuit = p_id_nuit
       AND type_evenement IN ('apnée obstructive','apnée centrale','RERA')),

    420,

    duree_hypoxie_min,

    position_dominante,

    (SELECT AVG(duree_sec)
     FROM evenement_respiratoire
     WHERE id_nuit = p_id_nuit),

    (SELECT MAX(duree_sec)
     FROM evenement_respiratoire
     WHERE id_nuit = p_id_nuit),

    decibels_max,
    decibels_moy,
    nb_ronflements_forts
);
END

Vue pour l'écriture du rapport, elle relie l'ID_patient aux résultats de la nuit_etude :

CREATE VIEW `resultat_nuit_lies_patient` AS
SELECT resultat_nuit.date_validation, resultat_nuit.iah, resultat_nuit.spo2_min, resultat_nuit.spo2_moy, resultat_nuit.spo2_mediane, resultat_nuit.nb_apnees, resultat_nuit.nb_hypopnees, resultat_nuit.nb_rera, resultat_nuit.nb_microeveils, resultat_nuit.duree_sommeil_min, resultat_nuit.duree_hypoxie_min, resultat_nuit.position_dominante, resultat_nuit.duree_apnee_moy_sec, resultat_nuit.duree_apnee_max_sec, resultat_nuit.decibels_max, resultat_nuit.decibels_moy, resultat_nuit.nb_ronflements_forts, resultat_nuit.severite_iah, resultat_nuit.commentaire_medical, nuit_etude.id_patient, patient.nom
FROM resultat_nuit 
LEFT JOIN nuit_etude ON resultat_nuit.id_nuit = nuit_etude.id_nuit
LEFT JOIN patient ON nuit_etude.id_patient = patient.id_patient
